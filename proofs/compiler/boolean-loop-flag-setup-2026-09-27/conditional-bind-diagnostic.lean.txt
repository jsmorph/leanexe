import LeanExe.Extract.ScalarFunc
namespace FlagSetupDiagnostic
def rangeBoolFlagSetupCondition (count seed : UInt64) : Id Bool := do
  let flag ← if count == 0 then pure (seed == 0) else pure (seed % 2 == 0)
  let value ← (do
    let mut a := seed
    for i in [:count.toNat] do
      if flag then a := a + i.toUInt64 + 1 else a := a + 3
    return a)
  return flag && value != seed

end FlagSetupDiagnostic
set_option pp.all true in
run_elab do
  let env ← Lean.getEnv
  let some info := env.find? `FlagSetupDiagnostic.rangeBoolFlagSetupCondition | throwError "missing"
  let some value := info.value? | throwError "missing body"
  Lean.logInfo m!"{value}"
